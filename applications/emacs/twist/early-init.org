* Decrease GC
#+begin_src emacs-lisp
  (setopt gc-cons-threshold (* 1024 1024 1024))
  (add-hook 'emacs-startup-hook
            (lambda ()
              (setopt gc-cons-threshold (* 16 1024 1024))))
#+end_src

* profiler
#+begin_src emacs-lisp :tangle no
  (require 'profiler)
  (profiler-start 'cpu)
  (add-hook 'emacs-startup-hook
            (lambda ()
              (message "Emacs initialized in %.3f sec with %d garbage collection(s)"
                       (float-time (time-subtract after-init-time before-init-time))
                       gcs-done)
              (profiler-report)
              (profiler-stop)))
#+end_src

* Toolbar/Menubar
#+begin_src emacs-lisp
  (menu-bar-mode -1)
