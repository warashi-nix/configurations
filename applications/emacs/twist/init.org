* setup.el
#+begin_src emacs-lisp
(eval-when-compile
  (require 'setup)
  (require 'cl-lib)

  ;; https://www.emacswiki.org/emacs/SetupEl#h5o-4
  (defmacro defsetup (name signature &rest body)
    "Shorthand for `setup-define'.
  NAME is the name of the local macro.  SIGNATURE is used as the
  argument list for FN.  If BODY starts with a string, use this as
  the value for :documentation.  Any following keywords are passed
  as OPTS to `setup-define'."
    (declare (debug defun))
    (let (opts)
      (when (stringp (car body))
        (setq opts (nconc (list :documentation (pop body))
                          opts)))
      (while (keywordp (car body))
        (let* ((prop (pop body))
               (val `',(pop body)))
          (setq opts (nconc (list prop val) opts))))
      `(setup-define ,name
         (cl-function (lambda ,signature ,@body))
         ,@opts)))

  (defsetup :package (package)
            "Fake installation of PACKAGE."
            :repeatable t
            :shorthand cadr)

  (defsetup :nixpkgs (nixpkgs)
            "Fake installation of packages from NIXPKGS for executables."
            :repeatable t)

  (defsetup :opt (&rest pairs)
            "Set the option NAME to VAL using `setopt'."
            :after-loaded t
            `(setopt ,@pairs))

  (defsetup :global-keymap (key command)
            "Globally bind KEY to COMMAND."
            :repeatable t
            `(keymap-global-set ,key ,command))

  (defsetup :keymap (key command)
            "Use `keymap-set' to define keybindings."
            :after-loaded t
            :repeatable t
            `(keymap-set ,(setup-get 'map) ,key ,command))

  (defsetup :defer (sec &optional func)
            "Call a function at N seconds after init."
            `(add-hook 'after-init-hook
                       (lambda nil
                         (run-with-timer ,sec nil
                                         ,(or func
                                              `(function ,(setup-get 'mode)))))))

  (defsetup :after-init (func &rest args)
            "Run func with ARGS after init."
            `(add-hook 'after-init-hook
                       ,(if args
                            `(lambda nil (,func ,@args))
                          `(function ,func)))))
#+end_src

* シェル/環境
** exec-path-from-shell
macOS の GUI から起動した Emacs は launchd 直下で動くため、nix-darwin や home-manager が設定する環境変数を受け取れない。darwin ではクリーンな環境で起動したログインシェルから環境変数の全量を取得してコピーする。それ以外の環境では従来どおり exec-path-from-shell を使う。

#+begin_src emacs-lisp
(setup (:package exec-path-from-shell)
  (:opt exec-path-from-shell-variables '("PATH" "MANPATH" "SSH_AUTH_SOCK" "SSH_AGENT_PID" "GPG_AGENT_INFO" "LANG" "LC_CTYPE" "NIX_SSL_CERT_FILE" "NIX_PATH")
        exec-path-from-shell-arguments nil)
  (defun my/login-shell-environment-entries ()
    "クリーンな環境で起動したログインシェルの環境変数を NAME=VALUE のリストで返す。"
    ;; nix-darwin や home-manager の環境構築は __NIX_DARWIN_SET_ENVIRONMENT_DONE
    ;; などのガードで二重実行を避けており、Emacs の環境を引き継いだシェルでは
    ;; スキップされて現在の環境がそのまま返ってくるため、env -i で空の環境から
    ;; 起動してガードを迂回する
    (with-temp-buffer
      (apply #'call-process "/usr/bin/env" nil '(t nil) nil
             "-i"
             (append
              (mapcan (lambda (name)
                        (when-let* ((value (getenv name)))
                          (list (concat name "=" value))))
                      '("HOME" "USER" "LOGNAME" "SHELL"))
              (list (or (getenv "SHELL") shell-file-name) "-l" "-c" "env -0")))
      (split-string (buffer-string) "\0" t)))
  (defun my/import-login-shell-environment ()
    "ログインシェルの環境変数を全てコピーし、PATH と exec-path を更新する。"
    ;; シェルセッション固有の状態を表す変数は GUI Emacs 側の実態と食い違うため取り込まない
    (let ((wrapper-path (split-string (or (getenv "PATH") "") path-separator t))
          (exclude '("_" "SHLVL" "PWD" "OLDPWD" "TERM" "TTY" "HOME" "USER" "LOGNAME" "SHELL")))
      (dolist (entry (my/login-shell-environment-entries))
        (when-let* ((pos (string-search "=" entry))
                    (name (substring entry 0 pos)))
          (when (and (string-match-p "\\`[A-Za-z_][A-Za-z0-9_]*\\'" name)
                     (not (member name exclude)))
            (setenv name (substring entry (1+ pos))))))
      ;; twist の wrapper が --prefix PATH で足した nix store のパスはログイン
      ;; シェル側に存在しないため、コピーで消えたエントリを前置し直す
      (let* ((new-path (split-string (or (getenv "PATH") "") path-separator t))
             (missing (seq-remove (lambda (dir) (member dir new-path)) wrapper-path)))
        (setenv "PATH" (string-join (append missing new-path) path-separator))
        (setq exec-path (append missing new-path (list exec-directory))))))
  (defun my/initialize-environment ()
    "darwin ではログインシェルの環境変数を全てコピーし、それ以外では exec-path-from-shell で初期化する。"
    (static-if (eq system-type 'darwin)
        (my/import-login-shell-environment)
      (exec-path-from-shell-initialize)))
  (:after-init my/initialize-environment))
#+end_src

*** TUI のカーソル色
nskk の nskk-cursor-update は set-cursor-color を呼ぶだけで TUI では何も
起きないため、OSC 12 で端末にカーソル色を通知する。

#+begin_src emacs-lisp
(defun my/nskk-tty-cursor-color ()
  (and (bound-and-true-p nskk-mode)
       (bound-and-true-p nskk-current-state)
       (fboundp 'nskk--cursor-with-color)
       (nskk--cursor-with-color (nskk-state-mode nskk-current-state))))

(defun my/nskk-tty-cursor-spec (color)
  ;; 端末は X11 の色名を解釈できないので rgb: 表記に変換する
  (let ((values (and color (tty-color-standard-values color))))
    (when values
      (apply #'format "rgb:%04x/%04x/%04x" values))))

(defun my/nskk-tty-cursor-apply ()
  (when (eq (framep (selected-frame)) t)
    (let* ((terminal (frame-terminal))
           (spec (my/nskk-tty-cursor-spec (my/nskk-tty-cursor-color))))
      (unless (equal spec (terminal-parameter terminal 'my/nskk-tty-cursor-spec))
        (set-terminal-parameter terminal 'my/nskk-tty-cursor-spec spec)
        (send-string-to-terminal
         (if spec (format "\e]12;%s\a" spec) "\e]112\a")
         terminal)))))

(defun my/nskk-tty-cursor-apply-safe ()
  (condition-case nil
      (my/nskk-tty-cursor-apply)
    (error nil)))

(defun my/nskk-tty-cursor-reset-all ()
  (dolist (terminal (terminal-list))
    (when (and (eq (terminal-live-p terminal) t)
               (terminal-parameter terminal 'my/nskk-tty-cursor-spec))
      (set-terminal-parameter terminal 'my/nskk-tty-cursor-spec nil)
      (ignore-errors (send-string-to-terminal "\e]112\a" terminal)))))

(defun my/nskk-tty-cursor-reset-terminal (terminal)
  (when (terminal-parameter terminal 'my/nskk-tty-cursor-spec)
    (ignore-errors (send-string-to-terminal "\e]112\a" terminal))))

(setup emacs
  (add-hook 'delete-terminal-functions #'my/nskk-tty-cursor-reset-terminal)
  (add-hook 'post-command-hook #'my/nskk-tty-cursor-apply-safe)
  (add-hook 'kill-emacs-hook #'my/nskk-tty-cursor-reset-all))
#+end_src

** brainium
[[https://github.com/Warashi/brainium][brainium]] (PKM リポジトリ) を対象にした org-node の設定。

同じディレクトリ構造 (notes/ inbox.org tasks.org journal/) の PKM を、この
home-manager を flake inputs として使う非公開マシンでも使うため、リポジトリの
場所と追加 capture テンプレートを custom.el から差し替えられるようにする。
custom.el は init 冒頭でロードされる。defcustom のデフォルト初期化
(custom-initialize-reset) は、束縛済みの値や customize の saved-value を
標準値より優先するため、custom.el 側は setopt か custom-set-variables
(customize UI での保存を含む) で設定するだけでよく、with-eval-after-load は不要。

#+begin_example
;; 非公開マシンの custom.el の例
(setopt my/pkm-directory "~/path/to/private-pkm/")
(setopt my/pkm-extra-capture-templates
        `(("w" "Work" entry
           (file ,(file-name-concat my/pkm-directory "work.org"))
           "* %?\n%U")))
#+end_example

#+begin_src emacs-lisp
(defcustom my/pkm-directory "~/ghq/github.com/Warashi/brainium/"
  "PKM リポジトリのルートディレクトリ。"
  :type 'directory
  :group 'org)

(defcustom my/pkm-extra-capture-templates nil
  "PKM 用にマシン固有で追加する `org-capture-templates' のエントリ。"
  :type 'sexp
  :group 'org)
#+end_src

新規ノートの作成先は notes/ (フラット配置) に固定する。ファイル名の slug は
title 由来のため、日本語 title の場合は英語 kebab-case 規約へ手動リネームで合わせる。

        org-node-file-directory-ask (file-name-concat my/pkm-directory "notes/"))

マシン固有のテンプレートは my/pkm-extra-capture-templates で後ろに追加する。

スマホからの capture は n8n の pending queue に溜まる。C-c o p で pull し、
inbox.org へ追記して orglint が通ってから ack する。org 書式を組み立てるのは
Emacs のこの 1 箇所だけで、n8n は text をそのまま預かる。

consumer token は pull と ack ができてしまうため、iOS ショートカットが持つ
producer token とは別のものを SOPS 経由でファイルに置き、その内容を
Authorization ヘッダーにそのまま使う。

#+begin_src emacs-lisp
;; url-request-* を let で効かせるにも、応答のバッファローカル変数を読むにも
;; url-http がロード済みである必要がある。
(require 'url-http)
(require 'iso8601)

;; url-http がバッファローカルに設定する変数。byte-compile に特殊変数だと
;; 教えるためだけの宣言。
(defvar url-http-response-status)
(defvar url-http-end-of-headers)

(defcustom my/pkm-mobile-capture-endpoint
  "https://n8n.warashi.dev/webhook/mobile-capture"
  "モバイル capture の queue を提供する n8n webhook のベース URL。"
  :type 'string
  :group 'org)

(defcustom my/pkm-mobile-capture-token-file
  (file-name-concat (or (getenv "XDG_CONFIG_HOME") "~/.config")
                    "mobile-capture/consumer-token")
  "consumer 用の Authorization ヘッダー値を収めたファイル。"
  :type 'file
  :group 'org)

(defcustom my/pkm-orglint-command
  (let ((dir (directory-file-name (expand-file-name my/pkm-directory))))
    (list "nix" "run" (concat dir "#orglint") "--" dir))
  "PKM に orglint をかけるコマンドと引数。"
  :type '(repeat string)
  :group 'org)

(defvar my/pkm--mobile-capture-unacked nil
  "ack しそこねた capture の id。次の同期の先頭で送り直す。")

(defun my/pkm--mobile-capture-request (method path &optional payload)
  "n8n の PATH へ METHOD で要求を出し、応答の JSON を alist で返す。
PAYLOAD が非 nil ならそれを JSON の body として送る。"
  (let* ((url-request-method method)
         (url-request-data
          (when payload (encode-coding-string (json-serialize payload) 'utf-8)))
         (url-request-extra-headers
          (append `(("Authorization"
                     . ,(with-temp-buffer
                          (insert-file-contents my/pkm-mobile-capture-token-file)
                          (string-trim (buffer-string)))))
                  (when payload '(("Content-Type" . "application/json")))))
         (buffer (url-retrieve-synchronously
                  (concat my/pkm-mobile-capture-endpoint path) t t 30)))
    (unless buffer
      (error "n8n への %s %s に失敗しました" method path))
    (unwind-protect
        (with-current-buffer buffer
          ;; ヘッダーは自分で探さない。Emacs 29 以降の url-http は CRLF を
          ;; 落とさなくなったため、"\n\n" を探す実装は本文に辿り着けない。
          (unless (and (<= 200 url-http-response-status)
                       (< url-http-response-status 300))
            (error "n8n が %d を返しました (%s %s)"
                   url-http-response-status method path))
          (goto-char url-http-end-of-headers)
          (decode-coding-region (point) (point-max) 'utf-8)
          (json-parse-buffer :object-type 'alist :array-type 'list))
      (kill-buffer buffer))))

(defun my/pkm--mobile-capture-to-org (capture &optional id)
  "CAPTURE を inbox.org のエントリ 1 件分の文字列にする。
CAPTURE は n8n の queue の 1 行で、text と createdAt を持つ alist。
ID を省略した場合は `org-id-new' で採番する。"
  (let* ((lines (split-string (string-trim (alist-get 'text capture)) "\n"))
         (body (string-trim-right (string-join (cdr lines) "\n"))))
    (concat "* " (car lines) "\n"
            ":PROPERTIES:\n"
            ":ID: " (or id (org-id-new)) "\n"
            ":CAPTURED: "
            (format-time-string
             (org-time-stamp-format t t)
             (encode-time (iso8601-parse (alist-get 'createdAt capture))))
            "\n"
            ":END:\n"
            (if (string-empty-p body) "" (concat body "\n")))))

(defun my/pkm--mobile-capture-ack ()
  "`my/pkm--mobile-capture-unacked' の id を n8n の queue から消す。"
  (when my/pkm--mobile-capture-unacked
    (my/pkm--mobile-capture-request
     "POST" "/ack" `((ids . ,(vconcat my/pkm--mobile-capture-unacked))))
    (setq my/pkm--mobile-capture-unacked nil)))

(defun my/pkm-sync-mobile-captures ()
  "n8n の queue にあるモバイル capture を inbox.org へ取り込む。"
  (interactive)
  (my/pkm--mobile-capture-ack)
  (let ((captures
         (alist-get 'items (my/pkm--mobile-capture-request "GET" "/pending"))))
    (if (null captures)
        (message "取り込む capture はありません")
      (find-file (file-name-concat my/pkm-directory "inbox.org"))
      (goto-char (point-max))
      (unless (bolp) (insert "\n"))
      (let ((start (point))
            (lint (get-buffer-create "*pkm-orglint*")))
        (dolist (capture captures)
          (insert (my/pkm--mobile-capture-to-org capture)))
        (save-buffer)
        (with-current-buffer lint (erase-buffer))
        (if (not (eq 0 (apply #'call-process (car my/pkm-orglint-command)
                              nil lint nil (cdr my/pkm-orglint-command))))
            ;; 追記を残したまま ack しないでおくと、次の同期で同じ capture が
            ;; 二重に入る。queue を唯一の控えに保つため取り込みごと巻き戻す。
            (progn
              (delete-region start (point-max))
              (save-buffer)
              (display-buffer lint)
              (error "orglint が通らなかったので取り込みを取り消しました"))
          (goto-char start)
          (setq my/pkm--mobile-capture-unacked
                (mapcar (lambda (capture) (alist-get 'id capture)) captures))
          (my/pkm--mobile-capture-ack)
          (message "%d 件取り込みました" (length captures)))))))

(keymap-global-set "C-c o p" #'my/pkm-sync-mobile-captures)
#+end_src

model と effort を固定した起動コマンドを用意する。eshell から呼べるようにするのは、
その時点の =default-directory= (TRAMP 先を含む) をそのまま引き継がせるため。

#+begin_src emacs-lisp
(defun warashi-agent-shell--apply-thought-level ()
  "agent config に載せた thought level (effort) を新しい shell に適用する。"
  ;; effort は agent-shell の agent config に設定点が無く、session 確立後に
  ;; ACP の config option として送るしかない。claude-agent-acp は初期 effort を
  ;; settings ファイルからしか読まないため、_meta 経由では指定できない。
  (when-let* ((config (alist-get :agent-config (agent-shell--state)))
              (level (alist-get :warashi-thought-level config)))
    (agent-shell-subscribe-to
     :shell-buffer (current-buffer)
     :event 'init-finished
     :on-event
     (lambda (_event)
       (agent-shell--config-option-set-thought-level-id
        :thought-level-id level
        :on-failure (lambda (acp-error _raw-message)
                      (message "Failed to set thought level %s: %s" level acp-error)))))))

(defun warashi-agent-shell--start-claude (model-id thought-level)
  "MODEL-ID と THOUGHT-LEVEL を指定して Claude agent-shell を起動する。"
  (require 'agent-shell-anthropic)
  (let ((config (agent-shell-anthropic-make-claude-code-config)))
    ;; :default-model-id は session 確立後に funcall されるので、動的束縛では
    ;; なく MODEL-ID を lexical に閉じ込めた関数へ差し替える。
    (setcdr (assq :default-model-id config) (lambda () model-id))
    ;; thought level を動的束縛で渡さないのは、`agent-shell-mode-hook' が
    ;; `agent-shell--dwim' の動的エクステント内で走るとは限らないため。
    ;; config は state の :agent-config に保存されるので、そこから読ませる。
    (push (cons :warashi-thought-level thought-level) config)
    (agent-shell--dwim :config config :new-shell t)))

(defmacro warashi-agent-shell-define-claude-variants (&rest variants)
  "VARIANTS から Claude agent-shell の起動コマンドを定義する。
VARIANTS の各要素は (NAME MODEL-ID THOUGHT-LEVEL)。NAME ごとに
`warashi-agent-shell-claude-NAME' と、eshell から短い名前で呼ぶための
`eshell/claude-NAME' を生成する。"
  ;; 一覧を defconst に置いてマクロから参照しないのは、byte-compile 時に
  ;; defconst が評価されず、マクロ展開時に void-variable になるため。
  `(progn
     ,@(mapcan
        (pcase-lambda (`(,name ,model-id ,thought-level))
          (let ((fn (intern (format "warashi-agent-shell-claude-%s" name)))
                (eshell-fn (intern (format "eshell/claude-%s" name))))
            (list
             `(defun ,fn ()
                ,(format "Claude agent-shell を model %s / effort %s で起動する。"
                         model-id thought-level)
                (interactive)
                (warashi-agent-shell--start-claude ,model-id ,thought-level))
             `(defun ,eshell-fn (&rest _args)
                ,(format "eshell から `%s' を起動する。" fn)
                (,fn)))))
        variants)))

;; MODEL-ID は agent-shell 起動時に "Available models" に出る ID。
;; sonnet-xhigh だけ 200k なのは、sonnet に 1M context 版の ID が無いため。
(warashi-agent-shell-define-claude-variants
 (sonnet-xhigh "sonnet"             "xhigh")
 (opus-low     "opus[1m]"           "low")
 (opus-medium  "opus[1m]"           "medium")
 (fable-medium "claude-fable-5[1m]" "medium"))

session の累積コストを context usage indicator の隣に常設する。ターン終了時の
ボックス (=agent-shell-show-usage-at-turn-end=) ではなく常設にするのは、走らせて

#+begin_src emacs-lisp
(defun warashi-agent-shell--cost-indicator ()
  "session の累積コストを表示用の文字列で返す。"
  ;; state を読むのに内部関数を使うのは、usage を取得する公開 API が無いため。
  (when-let* ((usage (map-elt (agent-shell--state) :usage))
              (amount (map-elt usage :cost-amount))
              ((> amount 0)))
    (let ((currency (map-elt usage :cost-currency)))
      ;; USD を $ に畳むのは、幅の限られる header で 2 文字を惜しむため。
      (format "%s%.2f" (if (member currency '(nil "USD")) "$" currency) amount))))

(defun warashi-agent-shell--append-cost-indicator (indicator)
  "context usage INDICATOR の後ろに cost を足す。"
  ;; mode-line ではなく context indicator に相乗りするのは、tty では
  ;; `agent-shell-header-style' が text になり、`agent-shell--mode-line-format'
  ;; が nil を返して情報が header-line 側にしか出ないため。
  ;; indicator が nil のときに cost だけ返さないのは、context 未取得の段階で
  ;; header に単独の数字が現れると何の値か分からないため。
  (if-let* ((indicator)
            (cost (warashi-agent-shell--cost-indicator)))
      (concat indicator " " cost)
    indicator))
