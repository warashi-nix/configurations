* setup.el
#+begin_src emacs-lisp
(eval-when-compile
  (require 'setup)
  (require 'cl-lib)

  ;; https://www.emacswiki.org/emacs/SetupEl#h5o-4
  (defmacro defsetup (name signature &rest body)
    "Shorthand for `setup-define'.
  NAME is the name of the local macro.  SIGNATURE is used as the
  argument list for FN.  If BODY starts with a string, use this as
  the value for :documentation.  Any following keywords are passed
  as OPTS to `setup-define'."
    (declare (debug defun))
    (let (opts)
      (when (stringp (car body))
        (setq opts (nconc (list :documentation (pop body))
                          opts)))
      (while (keywordp (car body))
        (let* ((prop (pop body))
               (val `',(pop body)))
          (setq opts (nconc (list prop val) opts))))
      `(setup-define ,name
         (cl-function (lambda ,signature ,@body))
         ,@opts)))

  (defsetup :package (package)
            "Fake installation of PACKAGE."
            :repeatable t
            :shorthand cadr)

  (defsetup :nixpkgs (nixpkgs)
            "Fake installation of packages from NIXPKGS for executables."
            :repeatable t)

  (defsetup :opt (&rest pairs)
            "Set the option NAME to VAL using `setopt'."
            :after-loaded t
            `(setopt ,@pairs))

  (defsetup :global-keymap (key command)
            "Globally bind KEY to COMMAND."
            :repeatable t
            `(keymap-global-set ,key ,command))

  (defsetup :keymap (key command)
            "Use `keymap-set' to define keybindings."
            :after-loaded t
            :repeatable t
            `(keymap-set ,(setup-get 'map) ,key ,command))

  (defsetup :defer (sec &optional func)
            "Call a function at N seconds after init."
            `(add-hook 'after-init-hook
                       (lambda nil
                         (run-with-timer ,sec nil
                                         ,(or func
                                              `(function ,(setup-get 'mode)))))))

  (defsetup :after-init (func &rest args)
            "Run func with ARGS after init."
            `(add-hook 'after-init-hook
                       ,(if args
                            `(lambda nil (,func ,@args))
                          `(function ,func)))))
#+end_src

* シェル/環境
** exec-path-from-shell
macOS の GUI から起動した Emacs は launchd 直下で動くため、nix-darwin や home-manager が設定する環境変数を受け取れない。darwin ではクリーンな環境で起動したログインシェルから環境変数の全量を取得してコピーする。それ以外の環境では従来どおり exec-path-from-shell を使う。

#+begin_src emacs-lisp
(setup (:package exec-path-from-shell)
  (:opt exec-path-from-shell-variables '("PATH" "MANPATH" "SSH_AUTH_SOCK" "SSH_AGENT_PID" "GPG_AGENT_INFO" "LANG" "LC_CTYPE" "NIX_SSL_CERT_FILE" "NIX_PATH")
        exec-path-from-shell-arguments nil)
  (defun my/login-shell-environment-entries ()
    "クリーンな環境で起動したログインシェルの環境変数を NAME=VALUE のリストで返す。"
    ;; nix-darwin や home-manager の環境構築は __NIX_DARWIN_SET_ENVIRONMENT_DONE
    ;; などのガードで二重実行を避けており、Emacs の環境を引き継いだシェルでは
    ;; スキップされて現在の環境がそのまま返ってくるため、env -i で空の環境から
    ;; 起動してガードを迂回する
    (with-temp-buffer
      (apply #'call-process "/usr/bin/env" nil '(t nil) nil
             "-i"
             (append
              (mapcan (lambda (name)
                        (when-let* ((value (getenv name)))
                          (list (concat name "=" value))))
                      '("HOME" "USER" "LOGNAME" "SHELL"))
              (list (or (getenv "SHELL") shell-file-name) "-l" "-c" "env -0")))
      (split-string (buffer-string) "\0" t)))
  (defun my/import-login-shell-environment ()
    "ログインシェルの環境変数を全てコピーし、PATH と exec-path を更新する。"
    ;; シェルセッション固有の状態を表す変数は GUI Emacs 側の実態と食い違うため取り込まない
    (let ((wrapper-path (split-string (or (getenv "PATH") "") path-separator t))
          (exclude '("_" "SHLVL" "PWD" "OLDPWD" "TERM" "TTY" "HOME" "USER" "LOGNAME" "SHELL")))
      (dolist (entry (my/login-shell-environment-entries))
        (when-let* ((pos (string-search "=" entry))
                    (name (substring entry 0 pos)))
          (when (and (string-match-p "\\`[A-Za-z_][A-Za-z0-9_]*\\'" name)
                     (not (member name exclude)))
            (setenv name (substring entry (1+ pos))))))
      ;; twist の wrapper が --prefix PATH で足した nix store のパスはログイン
      ;; シェル側に存在しないため、コピーで消えたエントリを前置し直す
      (let* ((new-path (split-string (or (getenv "PATH") "") path-separator t))
             (missing (seq-remove (lambda (dir) (member dir new-path)) wrapper-path)))
        (setenv "PATH" (string-join (append missing new-path) path-separator))
        (setq exec-path (append missing new-path (list exec-directory))))))
  (defun my/initialize-environment ()
    "darwin ではログインシェルの環境変数を全てコピーし、それ以外では exec-path-from-shell で初期化する。"
    (static-if (eq system-type 'darwin)
        (my/import-login-shell-environment)
      (exec-path-from-shell-initialize)))
  (:after-init my/initialize-environment))
#+end_src

** brainium
[[https://github.com/Warashi/brainium][brainium]] (PKM リポジトリ) を対象にした org-node の設定。

同じディレクトリ構造 (notes/ inbox.org tasks.org journal/) の PKM を、この
home-manager を flake inputs として使う非公開マシンでも使うため、リポジトリの
場所と追加 capture テンプレートを custom.el から差し替えられるようにする。
custom.el は init 冒頭でロードされる。defcustom のデフォルト初期化
(custom-initialize-reset) は、束縛済みの値や customize の saved-value を
標準値より優先するため、custom.el 側は setopt か custom-set-variables
(customize UI での保存を含む) で設定するだけでよく、with-eval-after-load は不要。

#+begin_example
;; 非公開マシンの custom.el の例
(setopt my/pkm-directory "~/path/to/private-pkm/")
(setopt my/pkm-extra-capture-templates
        `(("w" "Work" entry
           (file ,(file-name-concat my/pkm-directory "work.org"))
           "* %?\n%U")))
#+end_example

#+begin_src emacs-lisp
(defcustom my/pkm-directory "~/ghq/github.com/Warashi/brainium/"
  "PKM リポジトリのルートディレクトリ。"
  :type 'directory
  :group 'org)

(defcustom my/pkm-extra-capture-templates nil
  "PKM 用にマシン固有で追加する `org-capture-templates' のエントリ。"
  :type 'sexp
  :group 'org)
#+end_src

以降の設定はすべて my/pkm-directory から導出する。custom.el の値を確実に
反映するため即時の setopt で設定する。:opt はパッケージロード時まで評価が
遅延され、custom.el 側で導出値を直接上書きしたい場合に順序が絡むため使わない。

新規ノートの作成先は notes/ (フラット配置) に固定する。ファイル名の slug は
title 由来のため、日本語 title の場合は英語 kebab-case 規約へ手動リネームで合わせる。

#+begin_src emacs-lisp
(setopt org-mem-watch-dirs (list my/pkm-directory)
        org-node-file-directory-ask (file-name-concat my/pkm-directory "notes/"))
#+end_src

journal/<YYYY>/<YYYY-MM-DD>.org のデイリーノートを org-node-seq
(C-c o n s) で作成・移動する。新規デイリーにはファイルレベルの :ID: が付く。

#+begin_src emacs-lisp
;; org-node-seq-def-on-filepath-sort-by-basename は使わない。呼び出すと
;; init 評価時に org-node-seq のロードが必要になるうえ、:creator が
;; journal/<YYYY>/ の年サブディレクトリ配置を表現できないため、同ヘルパー
;; 相当の定義を書き下して :creator だけ差し替えている。
(setopt org-node-seq-defs
        (let ((dir (file-name-as-directory
                    (file-truename (file-name-concat my/pkm-directory "journal/")))))
          `(("d"
             :name "Journal"
             :version 2
             :classifier (lambda (node)
                           (when (string-prefix-p ,dir (org-mem-entry-file-truename node))
                             (let ((path (org-mem-entry-file node)))
                               (cons (file-name-base path) path))))
             :whereami (lambda ()
                         (when buffer-file-name
                           (let ((file-name (org-mem-file-known-p buffer-file-name)))
                             (when (and file-name (string-prefix-p ,dir file-name))
                               (file-name-base file-name)))))
             :prompter (lambda (key)
                         (let ((org-node-seq-that-marks-calendar key))
                           (org-read-date)))
             :try-goto (lambda (item)
                         (org-node-seq-try-goto-file (cdr item)))
             :creator (lambda (sortstr key)
                        (let ((org-node-creation-fn #'org-node-new-file)
                              (org-node-file-directory-ask
                               (file-name-concat ,dir (substring sortstr 0 4))))
                          (org-node-create sortstr (org-id-new) key)))))))
#+end_src

  ARGS supplies the output buffer and working directory."

              (or (plist-get args :buffer) justl--output-process-buffer))

             (default-directory directory)
             (buffer (ghostel-compile shell-command)))
        (if justl-pop-to-buffer-on-display
            (pop-to-buffer buffer)
          buffer)))

日英・英日の相互翻訳と、理由・言い換え候補つきの校正レビュー。
組み込みの =ellama-proofread= (C-c e P) は本文を直接置き換えるため、
講評を別バッファに出す非破壊のコマンドを別途用意する。

#+begin_src emacs-lisp
;; ellama 未ロードのままコンパイルされると let がレキシカル束縛になるため special 宣言する
(defvar ellama-language)

(setup ellama
  (:when-loaded
    (defvar warashi-ellama-proofread-prompt
      "あなたは経験豊富な編集者です。与えられた文章を校正してください。
原文が日本語なら日本語の文章として、英語なら英語の文章として校正します。
原文の文体・トーン・意図は尊重してください。
解説はすべて日本語で、次の構成で出力してください。

# 修正後の全文

修正をすべて反映した全文。

# 指摘事項

指摘ごとに以下を示す:
- 該当箇所: 原文からの抜粋
- 修正案: 修正後の表現
- 理由: なぜそう直すべきか
- 言い換え候補: ニュアンスの異なる表現を2〜3個、使い分けの説明つきで

# 総評

文章全体の良い点と改善の方向性。指摘すべき問題がない場合はその旨を述べる。"
      "校正レビュー用のシステムプロンプト。")

    (defun warashi-ellama--text ()
      "リージョンが有効ならその内容、なければバッファ全体を返す。"
      (if (region-active-p)
          (buffer-substring-no-properties (region-beginning) (region-end))
        (buffer-substring-no-properties (point-min) (point-max))))

    (defun warashi-ellama-translate ()
      "リージョンまたはバッファ全体を日英・英日で相互翻訳する。
日本語の文字を含むなら英語へ、含まないなら日本語へ翻訳する。"
      (interactive)
      (let ((ellama-language (if (string-match-p "\\cj" (warashi-ellama--text))
                                 "English"
                               "Japanese")))
        (if (region-active-p)
            (ellama-translate)
          (ellama-translate-buffer))))

    (defun warashi-ellama-proofread ()
      "リージョンまたはバッファ全体を校正する。
修正案とその理由、言い換え候補を別バッファに出力する。"
      (interactive)
      (ellama-instant (warashi-ellama--text)
                      :system warashi-ellama-proofread-prompt))))

(setup ellama-transient
  (:when-loaded
    (transient-append-suffix 'ellama-transient-main-menu "P"
      '("r" "Proofread with review" warashi-ellama-proofread))
    (transient-append-suffix 'ellama-transient-main-menu "d"
      '("l" "Translate (ja<->en)" warashi-ellama-translate))))

** ghostel
#+begin_src emacs-lisp
(setup (:package ghostel)
  (:also-load ghostel-compile
              ghostel-comint)
  (:opt ghostel-shell-integration nil
        ghostel-tramp-shell-integration nil
        ghostel-enable-osc52 t
        ghostel-module-directory (locate-user-emacs-file "ghostel-module"))
  (:after-init ghostel-compile-global-mode)
  (:after-init ghostel-comint-global-mode)

  (:when-loaded
    (add-to-list 'ghostel-tramp-shells '("rpc" login-shell))

    (defun warashi-envrc--ghostel-update-directory (update-directory directory)
      "Run UPDATE-DIRECTORY for DIRECTORY and refresh envrc when it changes."
      (let ((previous-directory default-directory))
        (prog1 (funcall update-directory directory)
          (unless (equal previous-directory default-directory)
            (cond
             ((file-remote-p default-directory)
              (when envrc-mode
                (envrc--apply (current-buffer) 'none)))
             (envrc-mode
              (envrc--update))
             ((and envrc-global-mode
                   (executable-find envrc-direnv-executable))
              (envrc-mode 1)))))))

    (unless (advice-member-p #'warashi-envrc--ghostel-update-directory
                             #'ghostel--update-directory)
      (advice-add #'ghostel--update-directory :around
                  #'warashi-envrc--ghostel-update-directory)))

  (add-to-list 'warning-suppress-types '(ghostel)))
#+end_src
