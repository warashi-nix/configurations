* setup.el
#+begin_src emacs-lisp
(eval-when-compile
  (require 'setup)
  (require 'cl-lib)

  ;; https://www.emacswiki.org/emacs/SetupEl#h5o-4
  (defmacro defsetup (name signature &rest body)
    "Shorthand for `setup-define'.
  NAME is the name of the local macro.  SIGNATURE is used as the
  argument list for FN.  If BODY starts with a string, use this as
  the value for :documentation.  Any following keywords are passed
  as OPTS to `setup-define'."
    (declare (debug defun))
    (let (opts)
      (when (stringp (car body))
        (setq opts (nconc (list :documentation (pop body))
                          opts)))
      (while (keywordp (car body))
        (let* ((prop (pop body))
               (val `',(pop body)))
          (setq opts (nconc (list prop val) opts))))
      `(setup-define ,name
         (cl-function (lambda ,signature ,@body))
         ,@opts)))

  (defsetup :package (package)
            "Fake installation of PACKAGE."
            :repeatable t
            :shorthand cadr)

  (defsetup :nixpkgs (nixpkgs)
            "Fake installation of packages from NIXPKGS for executables."
            :repeatable t)

  (defsetup :opt (&rest pairs)
            "Set the option NAME to VAL using `setopt'."
            :after-loaded t
            `(setopt ,@pairs))

  (defsetup :global-keymap (key command)
            "Globally bind KEY to COMMAND."
            :repeatable t
            `(keymap-global-set ,key ,command))

  (defsetup :keymap (key command)
            "Use `keymap-set' to define keybindings."
            :after-loaded t
            :repeatable t
            `(keymap-set ,(setup-get 'map) ,key ,command))

  (defsetup :defer (sec &optional func)
            "Call a function at N seconds after init."
            `(add-hook 'after-init-hook
                       (lambda nil
                         (run-with-timer ,sec nil
                                         ,(or func
                                              `(function ,(setup-get 'mode)))))))

  (defsetup :after-init (func &rest args)
            "Run func with ARGS after init."
            `(add-hook 'after-init-hook
                       ,(if args
                            `(lambda nil (,func ,@args))
                          `(function ,func)))))
#+end_src

* シェル/環境
** exec-path-from-shell
macOS の GUI から起動した Emacs は launchd 直下で動くため、nix-darwin や home-manager が設定する環境変数を受け取れない。darwin ではクリーンな環境で起動したログインシェルから環境変数の全量を取得してコピーする。それ以外の環境では従来どおり exec-path-from-shell を使う。

#+begin_src emacs-lisp
(setup (:package exec-path-from-shell)
  (:opt exec-path-from-shell-variables '("PATH" "MANPATH" "SSH_AUTH_SOCK" "SSH_AGENT_PID" "GPG_AGENT_INFO" "LANG" "LC_CTYPE" "NIX_SSL_CERT_FILE" "NIX_PATH")
        exec-path-from-shell-arguments nil)
  (defun my/login-shell-environment-entries ()
    "クリーンな環境で起動したログインシェルの環境変数を NAME=VALUE のリストで返す。"
    ;; nix-darwin や home-manager の環境構築は __NIX_DARWIN_SET_ENVIRONMENT_DONE
    ;; などのガードで二重実行を避けており、Emacs の環境を引き継いだシェルでは
    ;; スキップされて現在の環境がそのまま返ってくるため、env -i で空の環境から
    ;; 起動してガードを迂回する
    (with-temp-buffer
      (apply #'call-process "/usr/bin/env" nil '(t nil) nil
             "-i"
             (append
              (mapcan (lambda (name)
                        (when-let* ((value (getenv name)))
                          (list (concat name "=" value))))
                      '("HOME" "USER" "LOGNAME" "SHELL"))
              (list (or (getenv "SHELL") shell-file-name) "-l" "-c" "env -0")))
      (split-string (buffer-string) "\0" t)))
  (defun my/import-login-shell-environment ()
    "ログインシェルの環境変数を全てコピーし、PATH と exec-path を更新する。"
    ;; シェルセッション固有の状態を表す変数は GUI Emacs 側の実態と食い違うため取り込まない
    (let ((wrapper-path (split-string (or (getenv "PATH") "") path-separator t))
          (exclude '("_" "SHLVL" "PWD" "OLDPWD" "TERM" "TTY" "HOME" "USER" "LOGNAME" "SHELL")))
      (dolist (entry (my/login-shell-environment-entries))
        (when-let* ((pos (string-search "=" entry))
                    (name (substring entry 0 pos)))
          (when (and (string-match-p "\\`[A-Za-z_][A-Za-z0-9_]*\\'" name)
                     (not (member name exclude)))
            (setenv name (substring entry (1+ pos))))))
      ;; twist の wrapper が --prefix PATH で足した nix store のパスはログイン
      ;; シェル側に存在しないため、コピーで消えたエントリを前置し直す
      (let* ((new-path (split-string (or (getenv "PATH") "") path-separator t))
             (missing (seq-remove (lambda (dir) (member dir new-path)) wrapper-path)))
        (setenv "PATH" (string-join (append missing new-path) path-separator))
        (setq exec-path (append missing new-path (list exec-directory))))))
  (defun my/initialize-environment ()
    "darwin ではログインシェルの環境変数を全てコピーし、それ以外では exec-path-from-shell で初期化する。"
    (static-if (eq system-type 'darwin)
        (my/import-login-shell-environment)
      (exec-path-from-shell-initialize)))
  (:after-init my/initialize-environment))
#+end_src

*** 辞書読み込みとファイル名ハンドラ
nskk の辞書読み込みは、対象と全親ディレクトリにファイル名ハンドラが付い
ていないことを要求する (local-unhandled-file-p)。tramp の completion
handler は =/= や =/nix= にもマッチするので、tramp が読み込まれているだ
けで辞書も学習データも読めなくなる。しかも例外にはならず黙って ja-dic に
落ちるため、送りあり候補が出ないことでしか気付けない。

#+begin_src emacs-lisp
(defun my/nskk-read-without-file-handlers (fn &rest args)
  ;; 読み込みの間だけ外す。ハンドラを恒久的に触ると tramp 越しの編集が壊れる。
  (let ((file-name-handler-alist nil))
    (apply fn args)))

(dolist (fn '(nskk-dict-initialize
              nskk-search-load-learning-data
              nskk-study-load))
  (advice-add fn :around #'my/nskk-read-without-file-handlers))
#+end_src

*** 辞書の先読み
初回の =C-x C-j= は =nskk-dict-initialize= を同期で呼ぶ。SKK-JISYO.L
(17.5 万件) の読み込みは =dict-cache.eld= が効いていても数秒かかり、その
間 Emacs は固まる。daemon を上げ直すたびに最初の日本語入力で必ず踏む。

起動後のアイドルで一度 nskk-mode を通しておけば、初期化はすべて冪等ガー
ドに当たって以後の =C-x C-j= は即座に返る。

#+begin_src emacs-lisp
(defun my/nskk-preload-schedule ()
  (run-with-idle-timer 1 nil #'my/nskk-preload))

(defun my/nskk-preload ()
  ;; nskk-dict-initialize だけを先に呼ぶのでは駄目。辞書 fact が載った
  ;; prolog データベースに対して kana/converter の初期化が走ることになり、
  ;; 初回 nskk-mode は先読み無しの倍以上に伸びる。nskk-mode を一度通して
  ;; 本番と同じ順序で温める。
  (unless (eq 'done
              (while-no-input
                (require 'nskk)
                (with-temp-buffer
                  (nskk-mode 1)
                  (nskk-mode -1))
                'done))
    ;; 中断されたら次のアイドルでやり直す。ここで再スケジュールしないと、
    ;; 一度の打鍵で先読みが永久に諦められて元の重さに戻る。学習データの
    ;; 読み込み中に中断された場合だけは nskk--learning-loaded が先に立つ
    ;; ので retry では読み直されないが、当たる窓はミリ秒なので目を瞑る。
    (my/nskk-preload-schedule)))

(add-hook 'emacs-startup-hook #'my/nskk-preload-schedule)
#+end_src

*** マーカーを変更フックから隠さない
nskk は ▽ ▼ の挿入を inhibit-modification-hooks で隠す
(=nskk-henkan.el:600= の =nskk--insert-marker= ほか)。隠れた挿入でバッファ
の位置がずれるため、次の通常の挿入で track-changes の整合検査が落ちる。
=K= で ▽ が入った時点では無事で、続く =a= で かな が入った瞬間に死ぬ。

実害は copilot。after-change-functions でエラーが出ると Emacs はその関数
を hook から外すので、日本語を一文字打ったバッファでは以後 copilot が差分
を追わなくなる。copilot-mode は有効に見えたままなので気付けない。track-
changes の登録はバッファローカルなので他のバッファは巻き込まないが、日本
語を打つバッファから順に効かなくなる。eglot は Emacs 31 では track-changes
を使っていないので影響しない。

マーカーを普通に挿入すれば整合は保たれる。undo にマーカーが入るので
=C-x u= で ▼かんじ → ▽かんじ → ▽かん と打鍵を遡るようになるが、打った
手順を巻き戻しているだけで実害は無い。

実装は =warashi-nskk-marker= に置く。マーカーが変更フックに見えているかどうかは
バッファを見ても分からず、copilot が黙って効かなくなる形でしか出ないので、ert で
固定する。

#+begin_src emacs-lisp
(setup (:package warashi-nskk-marker)
  (with-eval-after-load 'nskk
    (require 'warashi-nskk-marker)
    (warashi-nskk-marker-install-advice)))
#+end_src

*** ▼ 変換中のセミコロンで次の見出しを始める
=;= の sticky shift は、▼ 変換中 (=nskk--sticky-shift-dispatch= の Arm 2)
だけ self-insert に落ちる。=;shokai= SPC =;kougi= SPC と打つと、二つ目の
=;= がそのまま挿入されて =初回;こうぎ= になる。大文字での =;shokai= SPC
=Kougi= は =nskk-self-insert= の implicit kakutei を通って 初回講義 になる
ので、=;= だけがこの経路から外れている。

▼ の =;= も大文字と同じ「確定してから次の見出しを開く」に合流させる。確
定するかどうかは自前で判定せず =nskk--implicit-kakutei-needed-p= に委ねる
ので、送り仮名の途中や候補一覧が出ている間は従来どおり本体の処理に流れる。

実装は =warashi-nskk-sticky= に置く。横取りする条件を間違えても例外にはならず、
=;;= でリテラルの =;= が出せなくなる、という別の打鍵が壊れる形で出るので、どちら
の分岐も ert で固定する。

#+begin_src emacs-lisp
(setup (:package warashi-nskk-sticky)
  (with-eval-after-load 'nskk
    (require 'warashi-nskk-sticky)
    (warashi-nskk-sticky-install-advice)))
#+end_src

*** カーソル色をバッファに追従させる
nskk の nskk-cursor-update は状態が変わったときにしか呼ばれず、色はフレー
ム単位で持たれる。かなモードのバッファから別のバッファへ移っても誰も更新
しないので、カーソルはかなの色のまま残る。TUI ではそもそも
set-cursor-color が効かない。どちらも post-command-hook でカレントバッファ
の状態に同期させ、TUI では OSC 12 で端末に通知する。

nskk 本体のカーソル制御 (nskk-use-color-cursor) は切ってある。復帰を担う
nskk--cursor-color-restore は、フレーム上に nskk バッファがひとつでも表示
されていると復帰を拒む (nskk--other-nskk-buffers-active-p) ため、分割ウィ
ンドウで片方にかなモードのバッファが見えているとこの同期が効かなくなる。
書き手をここ一本に絞る。

実装は =warashi-nskk-cursor= に置く。GUI の退避と復帰、TTY の OSC 送信は、間違え
ても「色が戻らない」「毎コマンド端末に書き続ける」という形でしか出ないので、ert
で固定する。

#+begin_src emacs-lisp
(setup (:package warashi-nskk-cursor)
  (:require warashi-nskk-cursor)
  (add-hook 'delete-terminal-functions #'warashi-nskk-cursor-tty-reset-terminal)
  (add-hook 'post-command-hook #'warashi-nskk-cursor-apply-safe)
  ;; post-command-hook はコマンド経由の切り替えしか拾わない。プロセス出力
  ;; などコマンド外で display-buffer が走る経路をこの 2 つで塞ぐ
  (add-hook 'window-buffer-change-functions #'warashi-nskk-cursor-apply-safe)
  (add-hook 'window-selection-change-functions #'warashi-nskk-cursor-apply-safe)
  (add-hook 'kill-emacs-hook #'warashi-nskk-cursor-tty-reset-all))
#+end_src

*** PoC: input-method-function でキーを先に受け取る
=nskk-mode-map= は打鍵の取り込みを =<remap> <self-insert-command>= 一本に頼って
いるが、remap は「そのキーが解決したコマンド」に引かれるため、major mode が印字
可能キーを直接バインドしていると nskk まで届かない。agent-shell の =n= がその例
で、=konnkai= が =こnnかい= になる。dired や magit でも同種の衝突が起きうる。

=input-method-function= はキーマップ探索より手前で呼ばれるので、ここで文字を
nskk 専用の合成イベントに差し替えてしまえば、どの major mode とも衝突しない。
quail のように自前の read ループは持たない。ループを持つとコマンドが走らなくな
り、=post-command-hook= に依存する corfu と capf が死ぬため。合成イベントを
=nskk-mode-map= 側で本来のハンドラに橋渡しする形なら、=this-command= が
=nskk-self-insert= のまま残るので corfu 連携はそのまま動く。

PoC として測りたいのは以下。=C-x C-j= 経路 (=current-input-method= は nil のま
ま) で発火するか、prefix key (=C-x n=) が壊れないか、=read-char= や
=y-or-n-p=、isearch、keyboard macro での挙動。=C-\= 経路との比較のために
=japanese-nskk= も登録しておく。

実装をここに置かず =warashi-nskk-im= に切り出しているのは、差し替えを見送る条件
と橋渡しの挙動を ert で固定したいため。init.org の中身は tangle されて初めてロー
ドされるので、テストから =require= できない。

#+begin_src emacs-lisp
(setup (:package warashi-nskk-im)
  (:require warashi-nskk-im)
  ;; C-\ 経路との比較のために japanese-nskk も登録しておく。
  (register-input-method
   "japanese-nskk" "Japanese" #'warashi-nskk-im-activate ""
   "SKK (nskk) via input-method-function")
  (:with-hook nskk-mode-hook
    (:hook warashi-nskk-im-setup)))
#+end_src

キーマップ側の辻褄合わせは =warashi-nskk-map= に置く。包む条件は
=warashi-nskk-im= の差し替え条件と同じものでなければならず、ずれると read-only
バッファで a は major mode に降りるのに q だけ nskk に吸われる、という形で出る。
両者が一致していることは ert で並べて確かめる。

  (with-eval-after-load 'nskk

    (warashi-nskk-im-install-dispatch-bindings))
  (:with-hook nskk-mode-hook
    (:hook warashi-nskk-map-restore-bound-commands)))
#+end_src

*** 単キーコマンドに入力メソッドを奪わせない
=agent-shell-mode-map= は =n= =p= =r= =+= =-= =0= を素のキーで握っており、プロン
プト上ではそれぞれのコマンドが =self-insert-command= を直接呼ぶ
(=agent-shell.el= の =agent-shell-next-item= ほか)。直接呼び出しは
=nskk-mode-map= の =<remap> <self-insert-command>= を通らないので、=n= が生の
ASCII のまま入り、続く =o= だけが nskk を通って =nお= になる。=konnkai= のように
=n= を含む読みが軒並み壊れる。

ddskk が無事なのは =skk-j-mode-map= に印字可能 ASCII を直接バインドしていて、
minor-mode map が major-mode map より先に引かれるため。nskk は remap 一本なので
=n= が先に =agent-shell-next-item= に解決され、以後 =[remap
agent-shell-next-item]= しか探されない。

実装は =warashi-agent-shell= に切り出している。remap の有無で挙動が変わるうえ、
壊れても例外にはならず「入力した文字が少しおかしい」としてしか出ないので、ert で
