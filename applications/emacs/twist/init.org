* setup.el
#+begin_src emacs-lisp
(eval-when-compile
  (require 'setup)
  (require 'cl-lib)

  ;; https://www.emacswiki.org/emacs/SetupEl#h5o-4
  (defmacro defsetup (name signature &rest body)
    "Shorthand for `setup-define'.
  NAME is the name of the local macro.  SIGNATURE is used as the
  argument list for FN.  If BODY starts with a string, use this as
  the value for :documentation.  Any following keywords are passed
  as OPTS to `setup-define'."
    (declare (debug defun))
    (let (opts)
      (when (stringp (car body))
        (setq opts (nconc (list :documentation (pop body))
                          opts)))
      (while (keywordp (car body))
        (let* ((prop (pop body))
               (val `',(pop body)))
          (setq opts (nconc (list prop val) opts))))
      `(setup-define ,name
         (cl-function (lambda ,signature ,@body))
         ,@opts)))

  (defsetup :package (package)
            "Fake installation of PACKAGE."
            :repeatable t
            :shorthand cadr)

  (defsetup :nixpkgs (nixpkgs)
            "Fake installation of packages from NIXPKGS for executables."
            :repeatable t)

  (defsetup :opt (&rest pairs)
            "Set the option NAME to VAL using `setopt'."
            :after-loaded t
            `(setopt ,@pairs))

  (defsetup :global-keymap (key command)
            "Globally bind KEY to COMMAND."
            :repeatable t
            `(keymap-global-set ,key ,command))

  (defsetup :keymap (key command)
            "Use `keymap-set' to define keybindings."
            :after-loaded t
            :repeatable t
            `(keymap-set ,(setup-get 'map) ,key ,command))

  (defsetup :defer (sec &optional func)
            "Call a function at N seconds after init."
            `(add-hook 'after-init-hook
                       (lambda nil
                         (run-with-timer ,sec nil
                                         ,(or func
                                              `(function ,(setup-get 'mode)))))))

  (defsetup :after-init (func &rest args)
            "Run func with ARGS after init."
            `(add-hook 'after-init-hook
                       ,(if args
                            `(lambda nil (,func ,@args))
                          `(function ,func)))))
#+end_src

(setup (:package justl)
  (:nixpkgs just)
  (:also-load ghostel-compile)
  (:when-loaded
    (require 'ghostel-compile)

    (defun justl-ghostel--make-process
        (_original-function command &optional args _parent-buffer)
      "Run justl COMMAND through `ghostel-compile'.
  ARGS supplies the output buffer and working directory."
      (let* ((buffer-name
              (or (plist-get args :buffer) justl--output-process-buffer))
             (directory
              (or (plist-get args :directory)
                  (and justl-justfile (file-name-directory justl-justfile))
                  default-directory))
             (shell-command (mapconcat #'shell-quote-argument command " "))
             (ghostel-compile-buffer-name buffer-name)
             (compile-command compile-command)
             (compilation-ask-about-save t)
             (compilation-save-buffers-predicate #'ignore)
             (default-directory directory)
             (buffer (ghostel-compile shell-command)))
        (if justl-pop-to-buffer-on-display
            (pop-to-buffer buffer)
          buffer)))

    (unless (advice-member-p #'justl-ghostel--make-process
                             'justl--make-process)
      (advice-add 'justl--make-process :around
                  #'justl-ghostel--make-process))))

** ghostel
#+begin_src emacs-lisp
(setup (:package ghostel)
  (:also-load ghostel-compile
              ghostel-comint)
  (:opt ghostel-shell-integration nil
        ghostel-tramp-shell-integration nil
        ghostel-enable-osc52 t
        ghostel-module-directory (locate-user-emacs-file "ghostel-module"))
  (:global-keymap "C-c s" #'ghostel)
  (:after-init ghostel-compile-global-mode)
  (:after-init ghostel-comint-global-mode)

  (:when-loaded
    (add-to-list 'ghostel-tramp-shells '("rpc" login-shell))

    (defun warashi-envrc--ghostel-update-directory (update-directory directory)
      "Run UPDATE-DIRECTORY for DIRECTORY and refresh envrc when it changes."
      (let ((previous-directory default-directory))
        (prog1 (funcall update-directory directory)
          (unless (equal previous-directory default-directory)
            (cond
             ((file-remote-p default-directory)
              (when envrc-mode
                (envrc--apply (current-buffer) 'none)))
             (envrc-mode
              (envrc--update))
             ((and envrc-global-mode
                   (executable-find envrc-direnv-executable))
              (envrc-mode 1)))))))

    (unless (advice-member-p #'warashi-envrc--ghostel-update-directory
                             #'ghostel--update-directory)
      (advice-add #'ghostel--update-directory :around
                  #'warashi-envrc--ghostel-update-directory)))

  (add-to-list 'warning-suppress-types '(ghostel)))
